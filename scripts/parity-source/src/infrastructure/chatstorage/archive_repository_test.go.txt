func TestArchiveCombinedFiltersAndStablePagination(t *testing.T){
 r:=newTestSQLiteRepository(t);ctx:=context.Background();now:=time.Date(2026,9,1,0,0,0,0,time.UTC)
 a,b:="111@s.whatsapp.net","222@s.whatsapp.net";chat,other:="10@g.us","20@g.us"
 seedChatMessage(t,r,a,chat,"a","invoice 100%",now)
 seedChatMessage(t,r,a,chat,"b","invoice 100%",now.Add(time.Minute))
 seedChatMessage(t,r,a,other,"b","invoice 100%",now.Add(time.Minute))
 seedChatMessage(t,r,a,other,"c","invoice 1000",now.Add(2*time.Minute))
 seedChatMessage(t,r,b,chat,"b","PRIVATE OTHER ACCOUNT",now.Add(time.Minute))
 _,err:=r.db.Exec("UPDATE messages SET media_type='image',is_from_me=1,sender=? WHERE device_id=? AND id='b'",a,a);require.NoError(t,err)
 f:=domain.ArchiveFilter{DeviceID:a,Search:"invoice",Limit:2,Asc:true}
 first,total,err:=r.QueryArchive(ctx,f);require.NoError(t,err);require.Equal(t,int64(4),total);require.Len(t,first,2)
 f.Offset=2;second,total,err:=r.QueryArchive(ctx,f);require.NoError(t,err);require.Equal(t,int64(4),total);require.Len(t,second,2)
 seen:=map[string]bool{}
 for _,m:=range append(first,second...){key:=m.ChatJID+"/"+m.ID;require.False(t,seen[key],"duplicate page row");seen[key]=true;require.Equal(t,a,m.DeviceID)}
 start,end:=now.Add(time.Minute),now.Add(time.Minute);sent:=true
 f=domain.ArchiveFilter{DeviceID:a,ChatJID:chat,Sender:a,Search:"100%",StartTime:&start,EndTime:&end,MediaOnly:true,MediaType:"image",MessageType:"image",IsFromMe:&sent,Limit:10}
 rows,total,err:=r.QueryArchive(ctx,f);require.NoError(t,err);require.Equal(t,int64(1),total);require.Len(t,rows,1);require.Equal(t,"b",rows[0].ID)
 f.Offset=1;rows,total,err=r.QueryArchive(ctx,f);require.NoError(t,err);require.Empty(t,rows);require.Equal(t,int64(1),total)
 f=domain.ArchiveFilter{DeviceID:a,Search:"100%",Limit:10};rows,total,err=r.QueryArchive(ctx,f);require.NoError(t,err);require.Len(t,rows,3);require.Equal(t,int64(3),total)
 f=domain.ArchiveFilter{DeviceID:a,MessageType:"text",Limit:10};rows,total,err=r.QueryArchive(ctx,f);require.NoError(t,err);require.Len(t,rows,2);require.Equal(t,int64(2),total)
}
func TestArchiveContextAndCoverageUseFullIdentity(t *testing.T){
 r:=newTestSQLiteRepository(t);ctx:=context.Background();now:=time.Date(2026,9,1,0,0,0,0,time.UTC)
 a,b:="111@s.whatsapp.net","222@s.whatsapp.net";chat:="10@g.us"
 for _,id:=range []string{"a","b","c","d"}{seedChatMessage(t,r,a,chat,id,id,now)}
 seedChatMessage(t,r,b,chat,"b","private",now)
 seedChatMessage(t,r,a,"other@g.us","b","other chat",now)
 _,err:=r.db.Exec("UPDATE messages SET media_type='call' WHERE device_id=? AND chat_jid=? AND id='a'",a,chat);require.NoError(t,err)
 _,err=r.db.Exec("UPDATE messages SET media_type='image' WHERE device_id=? AND chat_jid=? AND id='c'",a,chat);require.NoError(t,err)
 before,anchor,after,err:=r.ContextArchive(ctx,a,chat,"b",1,1);require.NoError(t,err);require.Equal(t,"b",anchor.Content);require.Len(t,before,1);require.Equal(t,"a",before[0].ID);require.Len(t,after,1);require.Equal(t,"c",after[0].ID)
 before,_,after,err=r.ContextArchive(ctx,a,chat,"b",0,0);require.NoError(t,err);require.Empty(t,before);require.Empty(t,after)
 _,_,_,err=r.ContextArchive(ctx,a,chat,"missing",1,1);require.ErrorIs(t,err,domain.ErrArchiveAnchorNotFound)
 coverage,err:=r.CoverageArchive(ctx,a,chat);require.NoError(t,err);require.Equal(t,int64(4),coverage.MessageCount);require.Equal(t,int64(1),coverage.MediaCount);require.Equal(t,"b",coverage.AnchorMessageID);require.NotNil(t,coverage.OldestTimestamp)
 coverage,err=r.CoverageArchive(ctx,a,"empty@g.us");require.NoError(t,err);require.Zero(t,coverage.MessageCount);require.Nil(t,coverage.OldestTimestamp);require.Empty(t,coverage.AnchorMessageID)
}
func TestArchiveReactionHydrationDoesNotMixChatIDs(t *testing.T){
 r:=newTestSQLiteRepository(t);ctx:=context.Background();now:=time.Now().UTC();device:="111@s.whatsapp.net"
 for _,chat:=range []string{"a@g.us","b@g.us"}{
  seedChatMessage(t,r,device,chat,"same-id","hello",now)
  require.NoError(t,r.StoreReaction(&domain.Reaction{DeviceID:device,ChatJID:chat,MessageID:"same-id",ReactorJID:"333@s.whatsapp.net",Emoji:chat,Timestamp:now}))
 }
 rows,_,err:=r.QueryArchive(ctx,domain.ArchiveFilter{DeviceID:device,Limit:10});require.NoError(t,err);require.Len(t,rows,2)
 for _,m:=range rows{require.Len(t,m.Reactions,1);require.Equal(t,m.ChatJID,m.Reactions[0].Emoji)}
}
func TestArchiveErrorsAreBoundedAndObservable(t *testing.T){
 r:=newTestSQLiteRepository(t);ctx:=context.Background()
 for _,f:=range []domain.ArchiveFilter{{Limit:1},{DeviceID:"a",Limit:0},{DeviceID:"a",Limit:501},{DeviceID:"a",Limit:1,Offset:-1},{DeviceID:"a",Limit:1,Offset:100001},{DeviceID:"a",Limit:1,MessageType:"arbitrary"}}{
  _,_,err:=r.QueryArchive(ctx,f);require.Error(t,err)
 }
 start,end:=time.Now(),time.Now().Add(-time.Hour)
 _,_,err:=r.QueryArchive(ctx,domain.ArchiveFilter{DeviceID:"a",Limit:1,StartTime:&start,EndTime:&end});require.Error(t,err)
 _,_,_,err=r.ContextArchive(ctx,"a","b","c",101,1);require.Error(t,err)
 cancelled,cancel:=context.WithCancel(ctx);cancel();_,_,err=r.QueryArchive(cancelled,domain.ArchiveFilter{DeviceID:"a",Limit:1});require.True(t,errors.Is(err,context.Canceled),err)
 require.NoError(t,r.db.Close());_,_,err=r.QueryArchive(ctx,domain.ArchiveFilter{DeviceID:"a",Limit:1});require.Error(t,err)
}
