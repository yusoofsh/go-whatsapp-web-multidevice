package usecase

import (
 "context"
 "errors"
 "fmt"
 "time"

 chat "github.com/aldinokemal/go-whatsapp-web-multidevice/domains/chat"
 storage "github.com/aldinokemal/go-whatsapp-web-multidevice/domains/chatstorage"
 "github.com/aldinokemal/go-whatsapp-web-multidevice/infrastructure/whatsapp"
)

var _ chat.IHistoryUsecase = (*serviceChat)(nil)

func historyMessages(messages []*storage.Message) []chat.HistoryMessage {
 out:=make([]chat.HistoryMessage,0,len(messages))
 for _,m:=range messages{
  if m==nil{continue};kind:=m.MediaType;if kind==""{kind="text"}
  out=append(out,chat.HistoryMessage{ID:m.ID,ChatJID:m.ChatJID,SenderJID:m.Sender,Content:m.Content,Timestamp:m.Timestamp.UTC().Format(time.RFC3339Nano),IsFromMe:m.IsFromMe,MediaType:m.MediaType,MessageType:kind,Filename:m.Filename,FileLength:m.FileLength})
 }
 return out
}
func parseHistoryTime(raw string)(*time.Time,error){
 if raw==""{return nil,nil}
 v,err:=time.Parse(time.RFC3339Nano,raw)
 if err!=nil{v,err=time.Parse("2006-01-02",raw)}
 if err!=nil{return nil,errors.New("time filter must be RFC3339 or YYYY-MM-DD")}
 return &v,nil
}
func (service serviceChat) History(ctx context.Context,req chat.HistoryRequest)(chat.HistoryResponse,error){
 out:=chat.HistoryResponse{Action:req.Action,Completeness:"unknown",BestEffort:true}
 device,ok:=whatsapp.DeviceFromContext(ctx)
 if !ok || device==nil || device.JID()==""{return out,errors.New("select a paired device for history access")}
 if service.chatStorageRepo==nil{return out,errors.New("chat storage is disabled")}
 ctx,cancel:=context.WithTimeout(ctx,15*time.Second);defer cancel()
 if req.Action=="request_backfill"{
  if req.Count<1 || req.Count>500{return out,errors.New("count must be 1..500")}
  res,err:=service.RequestChatHistory(ctx,chat.RequestChatHistoryRequest{ChatJID:req.ChatJID,Count:req.Count})
  if err!=nil{return out,err};out.Backfill=&res;out.Asynchronous=true;return out,nil
 }
 backend,ok:=service.chatStorageRepo.(storage.IArchiveRepository)
 if !ok{return out,errors.New("archive queries are unavailable for this storage backend")}
 switch req.Action{
 case "search_all","export":
  start,err:=parseHistoryTime(req.StartTime);if err!=nil{return out,err};end,err:=parseHistoryTime(req.EndTime);if err!=nil{return out,err}
  filter:=storage.ArchiveFilter{DeviceID:device.JID(),ChatJID:req.ChatJID,Sender:req.Sender,Search:req.Search,StartTime:start,EndTime:end,MediaOnly:req.MediaOnly,MediaType:req.MediaType,MessageType:req.MessageType,IsFromMe:req.IsFromMe,Limit:req.Limit,Offset:req.Offset,Asc:req.Action=="export"}
  if err=filter.Validate();err!=nil{return out,err}
  messages,total,err:=backend.QueryArchive(ctx,filter);if err!=nil{return out,err}
  out.Messages=historyMessages(messages);out.Limit=req.Limit;out.Offset=req.Offset;out.Total=total;out.NextOffset=req.Offset+len(messages);out.HasMore=int64(out.NextOffset)<total
  return out,nil
 case "context":
  before,anchor,after,err:=backend.ContextArchive(ctx,device.JID(),req.ChatJID,req.MessageID,req.Before,req.After);if err!=nil{return out,err}
  if anchor==nil{return out,storage.ErrArchiveAnchorNotFound}
  converted:=historyMessages([]*storage.Message{anchor});out.Anchor=&converted[0];out.Before=historyMessages(before);out.After=historyMessages(after);out.Total=int64(len(before)+1+len(after));return out,nil
 case "coverage":
  coverage,err:=backend.CoverageArchive(ctx,device.JID(),req.ChatJID);if err!=nil{return out,err};out.Coverage=&coverage;out.Total=coverage.MessageCount;return out,nil
 default:return out,fmt.Errorf("unknown history action: %s",req.Action)
 }
}
